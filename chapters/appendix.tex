\chapter{\ifenglish Manual\else คู่มือการติดตั้งระบบ\fi}

\begin{mypara}
    \indent ในภาคผนวกนี้จะเป็นคู่มือการติดตั้งระบบสำหรับผู้ที่สนใจนำระบบไปใช้งานต่อหรือพัฒนาต่อจากโครงงานนี้ โดยจะมีรายละเอียดเกี่ยวกับขั้นตอนการติดตั้งเบื้องต้นของระบบ โดยแบ่งเป็น 34 ส่วนประกอบหลักได้แก่
    ส่วนติดต่อผู้ใช้ (Frontend Display) ส่วนบริการการคำนวณ (Logical Model) ส่วนบริการการจัดการ (Backend Controller) และระบบฐานข้อมูล (Database)  ซึ่งรายละเอียดการเตรียมสภาพแวดล้อมพื้นฐาน 
    และการติดตั้งระบบทั้งหมดจะถูกอธิบายไว้ในส่วนนี้
\end{mypara}

\section{ภาพรวมของระบบ}
\begin{mypara}
    \indent ระบบ DeSS-T ประกอบด้วย 4 ส่วนหลัก ได้แก่
\end{mypara}
\begin{itemize}
    \item ส่วนติดต่อผู้ใช้ (Frontend Display): React + Vite + TypeScript
    \item ส่วนบริการการคำนวณ (Logical Model): Python
    \item ส่วนบริการการจัดการ (Backend Controller): Go
    \item ระบบฐานข้อมูล (Database): PostgreSQL พร้อมส่วนขยาย PostGIS
\end{itemize}
\begin{mypara}
    \indent โครงสร้างโฟลเดอร์หลักของโครงการมีดังนี้
\end{mypara}

\begin{verbatim}
DeSS_T/
|- docker-compose.yml
|- .env
|- README.md
|- DeSS_T_Frontend/
|- DeSS_T_Backend-go/
`- DeSS_T_Backend-python/
\end{verbatim}

\section{การกำหนดค่าตัวแปรแวดล้อม}
\begin{mypara}
    \indent ให้สร้างหรือแก้ไขไฟล์ \texttt{.env} ที่รากของโปรเจกต์ โดยกำหนดค่าอย่างน้อยดังนี้
\end{mypara}

\begin{verbatim}
# ORS
ORS_API_KEY=YOUR_ORS_API_KEY

# Database
DB_HOST=localhost
DB_PORT=5432
DB_USER=DeSS_T
DB_PASSWORD=CHANGE_ME
DB_NAME=dess_db
\end{verbatim}

นอกจากนี้ ในโฟลเดอร์ Frontend ต้องมีไฟล์ \texttt{.env} (เช่น \texttt{DeSS\_T\_Frontend/.env}) และกำหนดค่าอย่างน้อยดังนี้
\begin{verbatim}
VITE_GOOGLE_CLIENT_ID=YOUR_GOOGLE_CLIENT_ID
\end{verbatim}

หมายเหตุ: ตัวแปรที่ขึ้นต้นด้วย \texttt{VITE\_} เป็นรูปแบบที่ Vite ใช้สำหรับเปิดให้ฝั่ง Frontend เข้าถึงค่าจาก environment ได้

ข้อควรระวังสำคัญมีดังนี้
\begin{enumerate}
    \item ไม่ควรบันทึกค่าความลับจริง (เช่น API key หรือรหัสผ่านจริง) ลงใน public repository
    \item ควรเปลี่ยนรหัสผ่านเริ่มต้นก่อนนำไปใช้งานจริง
\end{enumerate}

\section{การตั้งค่า PostgreSQL และ PostGIS}
\begin{mypara}
    \indent เนื่องจากระบบต้องพึ่งพาความสามารถเชิงพื้นที่ (Spatial) จึงจำเป็นต้องเปิดใช้งาน PostGIS ในฐานข้อมูลก่อนเริ่มบริการ backend
\end{mypara}

\subsection{ทางเลือก A: ติดตั้ง PostgreSQL ภายในเครื่อง (Manual)}
\begin{mypara}
    \indent เชื่อมต่อ PostgreSQL ด้วยสิทธิ์ผู้ดูแลระบบ และรันคำสั่ง
\end{mypara}
\begin{verbatim}
CREATE USER "DeSS_T" WITH PASSWORD 'CHANGE_ME';
CREATE DATABASE dess_db OWNER "DeSS_T";
\c dess_db
CREATE EXTENSION IF NOT EXISTS postgis;
CREATE EXTENSION IF NOT EXISTS postgis_topology;
\end{verbatim}

ตรวจสอบการทำงานของ PostGIS
\begin{verbatim}
SELECT PostGIS_Version();
\end{verbatim}

\subsection{ทางเลือก B: ใช้ PostgreSQL + PostGIS ผ่าน Docker}
\begin{mypara}
    \indent สำหรับการทดสอบแบบรวดเร็ว สามารถสร้างคอนเทนเนอร์ด้วยคำสั่ง
\end{mypara}

\begin{verbatim}
docker run --name dess-postgis -e POSTGRES_USER=DeSS_T -e POSTGRES_PASSWORD=CHANGE_ME -e POSTGRES_DB=dess_db -p 5432:5432 -d postgis/postgis:15-3.4
\end{verbatim}

และตรวจสอบเวอร์ชัน PostGIS
\begin{verbatim}
docker exec -it dess-postgis psql -U DeSS_T -d dess_db -c "SELECT PostGIS_Version();"
\end{verbatim}

\section{แนวทางตรวจสอบหลังติดตั้ง (Health Check)}
หลังจากติดตั้งและเริ่มระบบแล้ว ควรตรวจสอบรายการต่อไปนี้
\begin{enumerate}
    \item PostgreSQL สามารถเชื่อมต่อได้ตาม host/port ที่กำหนด
    \item คำสั่ง \texttt{SELECT PostGIS\_Version();} คืนค่าเวอร์ชันได้ถูกต้อง
    \item Go backend เริ่มทำงานได้โดยไม่มีข้อผิดพลาดการเชื่อมต่อฐานข้อมูล
    \item Python backend เริ่มทำงานได้โดยไม่มีปัญหา import หรือปัญหาฐานข้อมูล
    \item Frontend สามารถเรียกใช้งาน API ของ backend ได้สำเร็จ
\end{enumerate}

\section{ปัญหาที่พบบ่อยและแนวทางแก้ไข}
\begin{enumerate}
    \item \textbf{พอร์ตถูกใช้งานอยู่แล้ว} \\
    เปลี่ยนพอร์ตของบริการ หรือหยุดโปรเซสที่ใช้พอร์ตเดิม

    \item \textbf{ยืนยันตัวตนฐานข้อมูลไม่ผ่าน} \\
    ตรวจสอบค่า \texttt{DB\_USER}, \texttt{DB\_PASSWORD}, \texttt{DB\_NAME} และสิทธิ์ของผู้ใช้ฐานข้อมูล

    \item \textbf{ไม่พบฟังก์ชันของ PostGIS} \\
    ตรวจสอบว่าได้รัน \texttt{CREATE EXTENSION postgis;} ในฐานข้อมูลที่ถูกต้อง

    \item \textbf{เกิดปัญหา dependency ฝั่ง Frontend} \\
    ลบและติดตั้งแพ็กเกจใหม่
\begin{verbatim}
Remove-Item -Recurse -Force .\DeSS_T_Frontend\node_modules
npm install --prefix .\DeSS_T_Frontend
\end{verbatim}

    \item \textbf{เกิดปัญหาแพ็กเกจฝั่ง Python} \\
    ตรวจสอบว่าได้ activate virtual environment ก่อนติดตั้งแพ็กเกจและรันแอปพลิเคชัน
\end{enumerate}

\section{ข้อพิจารณาด้านความปลอดภัย}
\begin{enumerate}
    \item ควรเปลี่ยน (rotate) API key และรหัสผ่านฐานข้อมูลอย่างสม่ำเสมอ
    \item ควรป้องกันไฟล์ \texttt{.env} ไม่ให้รั่วไหลสู่ระบบควบคุมเวอร์ชันในสภาพแวดล้อมที่มีข้อมูลสำคัญ
    \item ควรแยกชุดข้อมูลรับรอง (credentials) สำหรับ dev, staging และ production ออกจากกันอย่างชัดเจน
\end{enumerate}
