\chapter{\ifenglish Manual\else คู่มือการติดตั้งระบบ\fi}

\begin{mypara}
    \indent ในภาคผนวกนี้จะเป็นคู่มือการติดตั้งระบบสำหรับผู้ที่สนใจนำระบบไปใช้งานต่อหรือพัฒนาต่อจากโครงงานนี้ โดยจะมีรายละเอียดเกี่ยวกับขั้นตอนการติดตั้งเบื้องต้นของระบบ โดยแบ่งเป็น 34 ส่วนประกอบหลักได้แก่
    ส่วนติดต่อผู้ใช้ (Frontend Display) ส่วนบริการการคำนวณ (Logical Model) ส่วนบริการการจัดการ (Backend Controller) และระบบฐานข้อมูล (Database)  ซึ่งรายละเอียดการเตรียมสภาพแวดล้อมพื้นฐาน 
    และการติดตั้งระบบทั้งหมดจะถูกอธิบายไว้ในส่วนนี้
\end{mypara}

\section{ภาพรวมของระบบ}
\begin{mypara}
    \indent ระบบ DeSS-T ประกอบด้วย 4 ส่วนหลัก ได้แก่
\end{mypara}
\begin{itemize}
    \item ส่วนติดต่อผู้ใช้ (Frontend Display): React + Vite + TypeScript
    \item ส่วนบริการการคำนวณ (Logical Model): Python
    \item ส่วนบริการการจัดการ (Backend Controller): Go
    \item ระบบฐานข้อมูล (Database): PostgreSQL พร้อมส่วนขยาย PostGIS
\end{itemize}
\begin{mypara}
    \indent โครงสร้างโฟลเดอร์หลักของโครงการมีดังนี้
\end{mypara}

\begin{verbatim}
DeSS_T/
|- docker-compose.yml
|- .env
|- README.md
|- DeSS_T_Frontend/
|- DeSS_T_Backend-go/
`- DeSS_T_Backend-python/
\end{verbatim}

\section{การตั้งค่า PostgreSQL และ PostGIS}
\begin{mypara}
    \indent เนื่องจากระบบต้องพึ่งพาความสามารถเชิงพื้นที่ (Spatial) จึงจำเป็นต้องเปิดใช้งาน PostGIS ในฐานข้อมูลก่อนเริ่มบริการ backend
\end{mypara}

\subsection{ทางเลือก A: ติดตั้ง PostgreSQL ภายในเครื่อง (Manual)}
\begin{mypara}
    \indent เชื่อมต่อ PostgreSQL ด้วยสิทธิ์ผู้ดูแลระบบ และรันคำสั่ง
\end{mypara}
\begin{verbatim}
CREATE USER "DeSS_T" WITH PASSWORD 'CHANGE_ME';
CREATE DATABASE dess_db OWNER "DeSS_T";
\c dess_db
CREATE EXTENSION IF NOT EXISTS postgis;
CREATE EXTENSION IF NOT EXISTS postgis_topology;
\end{verbatim}

ตรวจสอบการทำงานของ PostGIS
\begin{verbatim}
SELECT PostGIS_Version();
\end{verbatim}

\subsection{ทางเลือก B: ใช้ PostgreSQL + PostGIS ผ่าน Docker}
\begin{mypara}
    \indent สำหรับการทดสอบแบบรวดเร็ว สามารถสร้างคอนเทนเนอร์ด้วยคำสั่ง
\end{mypara}

\begin{verbatim}
docker run --name dess-postgis -e POSTGRES_USER=DeSS_T -e 
POSTGRES_PASSWORD=CHANGE_ME -e POSTGRES_DB=dess_db -p 5432:5432 -d 
postgis/postgis:15-3.4
\end{verbatim}

และตรวจสอบเวอร์ชัน PostGIS
\begin{verbatim}
docker exec -it dess-postgis psql -U DeSS_T -d dess_db -c 
"SELECT PostGIS_Version();"
\end{verbatim}

\section{แนวทางตรวจสอบหลังติดตั้ง (Health Check)}
หลังจากติดตั้งและเริ่มระบบแล้ว ควรตรวจสอบรายการต่อไปนี้
\begin{enumerate}
    \item PostgreSQL สามารถเชื่อมต่อได้ตาม host/port ที่กำหนด
    \item คำสั่ง \texttt{SELECT PostGIS\_Version();} คืนค่าเวอร์ชันได้ถูกต้อง
    \item Go backend เริ่มทำงานได้โดยไม่มีข้อผิดพลาดการเชื่อมต่อฐานข้อมูล
    \item Python backend เริ่มทำงานได้โดยไม่มีปัญหา import หรือปัญหาฐานข้อมูล
    \item Frontend สามารถเรียกใช้งาน API ของ backend ได้สำเร็จ
\end{enumerate}

\section{แนวทางการ deploy production บน VM}
ก่อนเริ่มการติดตั้ง ให้ดำเนินการเตรียมสภาพแวดล้อมบน VM ดังนี้:
\begin{itemize}
    \item \textbf{ติดตั้งเครื่องมือหลัก:} ติดตั้ง Docker Engine และ Docker Compose plugin
    \item \textbf{ตั้งค่า Firewall / Security Group:} เปิดพอร์ตที่จำเป็นสำหรับการสื่อสาร
    \begin{itemize}
        \item \texttt{80/443}: สำหรับ Reverse Proxy และการทำ SSL
        \item \texttt{3000}: สำหรับการเข้าถึง Frontend โดยตรง (ถ้าจำเป็น)
        \item \texttt{8080}: สำหรับ API Backend
        \item \textit{หมายเหตุ:} ไม่ควรเปิดพอร์ต \texttt{5432} (PostgreSQL) เป็นสาธารณะ
    \end{itemize}
    \item \textbf{สร้างโฟลเดอร์โปรเจกต์:} เช่น \texttt{/opt/dess-t} สำหรับเก็บไฟล์ Config ทั้งหมด
\end{itemize}

\subsection{จัดวางไฟล์สำคัญในโฟลเดอร์โปรเจกต์ที่สร้างขึ้น}
\begin{enumerate}
    \item \texttt{docker-compose.yml}
    \item \texttt{nginx.conf} (สำหรับการทำ mount เข้าไปยัง Frontend container)
    \item \texttt{.env} สำหรับสภาวะ Production โดยต้องระบุค่าที่สำคัญ ดังนี้:

\begin{verbatim}
ORS_API_KEY=your_key_here
DB_USER=admin
DB_PASSWORD=your_secure_password
DB_NAME=dess_t_db
\end{verbatim}
    \item ส่วน .env ที่ใช้สำหรับการตั้งค่า Google OAuth จำเป็นต้อง build พร้อม Image ของ Frontend ใหม่ทุกครั้งที่มีการเปลี่ยนแปลงค่าในไฟล์นี้ เนื่องจาก Vite จะทำการฝังค่าจาก environment ลงในโค้ดของ Frontend ในระหว่างขั้นตอนการ build เท่านั้น
\end{enumerate}
\subsection{เพื่อให้ระบบมีความปลอดภัยและเสถียร}
ควรปรับแก้ไฟล์ \texttt{docker-compose.yml} ดังนี้:
\begin{itemize}
    \item \textbf{Restart Policy:} ใส่ \texttt{restart: unless-stopped} ให้กับทุก Service
    \item \textbf{Port Binding:} 
    \begin{itemize}
        \item นำพอร์ต \texttt{5432:5432} ออก หรือ Bind เฉพาะ \texttt{127.0.0.1}
        \item พิจารณานำพอร์ต \texttt{5000} และ \texttt{8080} ออกหากต้องการให้ผ่าน Nginx เพียงทางเดียว
    \end{itemize}
    \item \textbf{Image Tagging:} ใช้ Tag แบบระบุเวอร์ชันที่แน่นอน (Immutable tags) เช่น \texttt{8.1} หรือ \texttt{8.4} แทนการใช้ \texttt{latest}
\end{itemize}

\subsection{การจัดการฐานข้อมูลครั้งแรก}
\begin{itemize}
    \item ตรวจสอบจนมั่นใจว่า PostgreSQL อยู่ในสถานะ \texttt{healthy}
    \item รันกระบวนการ Migration หรือ Seed ข้อมูล (หากมี)
    \item ทดสอบการเชื่อมต่อจาก Backend-Go ไปยัง Database
\end{itemize}

\subsection{การตั้งค่าโดเมนและ HTTPS}
\begin{itemize}
    \item ติดตั้ง Reverse Proxy ชั้นนอก (เช่น Nginx, Caddy หรือ Traefik) เพื่อรับ Traffic ที่พอร์ต 80/443
    \item ตั้งค่า SSL Certificate ผ่าน Let’s Encrypt
    \item ตรวจสอบ Google OAuth Redirect URI ให้ตรงกับ Domain ของ Production
\end{itemize}

\subsection{การเสริมความปลอดภัย (Hardening)}
\begin{itemize}
    \item \textbf{Secret Management:} ห้ามเก็บไฟล์ \texttt{.env} จริงไว้ในระบบ Version Control (Git)
    \item \textbf{Backup:} ตั้งค่าการสำรองข้อมูล Docker Volume อย่างสม่ำเสมอ
    \item \textbf{Log Rotation:} ตั้งค่าเพื่อป้องกันไม่ให้ Log กินพื้นที่ Disk จนเต็ม
    \item \textbf{Access Control:} จำกัดสิทธิ์ VM User และใช้ SSH Key แทน Password
\end{itemize}

\subsection{เมื่อมีการอัปเดต Source Code หรือเวอร์ชันใหม่}
\begin{verbatim}
# 1. แก้ไขเลข Tag ใน docker-compose.yml

docker compose up -d --pull always
\end{verbatim}

\section{ปัญหาที่พบบ่อยและแนวทางแก้ไข}
\begin{enumerate}
    \item \textbf{พอร์ตถูกใช้งานอยู่แล้ว} \\
    เปลี่ยนพอร์ตของบริการ หรือหยุดโปรเซสที่ใช้พอร์ตเดิม

    \item \textbf{ยืนยันตัวตนฐานข้อมูลไม่ผ่าน} \\
    ตรวจสอบค่า \texttt{DB\_USER}, \texttt{DB\_PASSWORD}, \texttt{DB\_NAME} และสิทธิ์ของผู้ใช้ฐานข้อมูล

    \item \textbf{ไม่พบฟังก์ชันของ PostGIS} \\
    ตรวจสอบว่าได้รัน \texttt{CREATE EXTENSION postgis;} ในฐานข้อมูลที่ถูกต้อง

    \item \textbf{เกิดปัญหา dependency ฝั่ง Frontend} \\
    ลบและติดตั้งแพ็กเกจใหม่
\begin{verbatim}
Remove-Item -Recurse -Force .\DeSS_T_Frontend\node_modules

\section{ข้อพิจารณาด้านความปลอดภัย}
\begin{enumerate}
    \item ควรเปลี่ยน (rotate) API key และรหัสผ่านฐานข้อมูลอย่างสม่ำเสมอ
    \item ควรป้องกันไฟล์ \texttt{.env} ไม่ให้รั่วไหลสู่ระบบควบคุมเวอร์ชันในสภาพแวดล้อมที่มีข้อมูลสำคัญ
    \item ควรแยกชุดข้อมูลรับรอง (credentials) สำหรับ dev, staging และ production ออกจากกันอย่างชัดเจน
\end{enumerate}
